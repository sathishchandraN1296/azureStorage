/* =========================================

========================================= */

const STORAGE_ACCOUNT = "marketingcloudreceipts";

const CONTAINER_NAME = "recepits";

const SAS_TOKEN = "sv=2026-02-06&ss=bfqt&srt=sco&sp=rwlacupitfx&se=2030-05-22T21:00:25Z&st=2026-05-22T12:45:25Z&spr=https&sig=lrWz2cOxFwogL%2Fy9obRZhkHc5eMR%2FBjiYxLU1llijPg%3D";

/* =========================================
   GLOBAL VARIABLES
========================================= */

let receiptCounter = 0;

/* =========================================
   ADD RECEIPT BLOCK
========================================= */

function addReceipt() {

    receiptCounter++;

    const block = document.createElement('div');

    block.className = 'receipt-block';

    block.innerHTML = `

        <h4>Receipt #${receiptCounter}</h4>

        <div class="row">

            <div class="col">

                <input type="date"
                       class="receipt-date"
                       required>

            </div>

            <div class="col">

                <input type="time"
                       class="receipt-time"
                       required>

            </div>

        </div>

        <div class="row">

            <div class="col">

                <input type="number"
                       class="receipt-amount"
                       placeholder="Receipt Amount"
                       step="0.01"
                       required>

            </div>

            <div class="col">

                <input type="text"
                       class="store-name"
                       placeholder="Store Name"
                       required>

            </div>

        </div>

        <div class="row">

            <div class="col">

                <input type="file"
                       accept=".jpg,.jpeg,.png,.pdf"
                       onchange="uploadFile(this)"
                       required>

            </div>

        </div>

        <div class="upload-url"></div>

        <input type="hidden"
               class="file-url">

    `;

    document
        .getElementById('receiptContainer')
        .appendChild(block);

    block
        .querySelector('.receipt-amount')
        .addEventListener('input', updateSummary);

    updateSummary();

}

/* =========================================
   UPLOAD FILE TO AZURE
========================================= */

async function uploadFile(input) {

    const file = input.files[0];

    if (!file) return;

    const block = input.closest('.receipt-block');

    try {

        /* =========================
           SHOW LOADING
        ========================= */

        block.querySelector('.upload-url').innerHTML = `
            <span style="color:orange;font-weight:bold;">
                Uploading file...
            </span>
        `;

        /* =========================
           FILE VALIDATION
        ========================= */

        const allowedTypes = [
            'image/jpeg',
            'image/png',
            'application/pdf'
        ];

        if (!allowedTypes.includes(file.type)) {

            alert('Only JPG, PNG and PDF files allowed');

            return;

        }

        /* =========================
           FILE SIZE VALIDATION
        ========================= */

        if (file.size > 5000000) {

            alert('Maximum file size is 5MB');

            return;

        }

        /* =========================
           UNIQUE FILE NAME
        ========================= */

        const uniqueFileName =
            Date.now() + "_" + file.name.replace(/\s/g, "_");

        /* =========================
           AZURE UPLOAD URL
        ========================= */

        const uploadURL =
        `https://${STORAGE_ACCOUNT}.blob.core.windows.net/${CONTAINER_NAME}/${uniqueFileName}?${SAS_TOKEN}`;

        console.log("Upload URL:", uploadURL);

        /* =========================
           UPLOAD FILE
        ========================= */

        const response = await fetch(uploadURL, {

            method: 'PUT',

            headers: {

                'x-ms-blob-type': 'BlockBlob',
                'Content-Type': file.type

            },

            body: file

        });

        console.log("Azure Upload Status:", response.status);

        if (!response.ok) {

            const errorText = await response.text();

            console.log(errorText);

            throw new Error("Azure upload failed");

        }

        /* =========================
           FINAL PUBLIC FILE URL
        ========================= */

        const finalFileURL =
        `https://${STORAGE_ACCOUNT}.blob.core.windows.net/${CONTAINER_NAME}/${uniqueFileName}`;

        console.log("Final File URL:", finalFileURL);

        /* =========================
           SAVE URL
        ========================= */

        block.querySelector('.file-url').value =
            finalFileURL;

        /* =========================
           DISPLAY URL
        ========================= */

        block.querySelector('.upload-url').innerHTML = `

            <span style="color:green;font-weight:bold;">
                File Uploaded Successfully
            </span>

            <br><br>

            <a href="${finalFileURL}"
               target="_blank"
               style="color:blue;">

               View Uploaded File

            </a>

        `;

    }
    catch(error) {

        console.error(error);

        block.querySelector('.upload-url').innerHTML = `

            <span style="color:red;font-weight:bold;">
                Upload Failed
            </span>

        `;

        alert('Azure upload failed');

    }

}

/* =========================================
   UPDATE SUMMARY
========================================= */

function updateSummary() {

    let total = 0;

    const receipts =

        document.querySelectorAll('.receipt-block');

    receipts.forEach(block => {

        total += parseFloat(

            block.querySelector('.receipt-amount').value || 0

        );

    });

    document.getElementById('displayTotal')
        .innerText = total.toFixed(2);

    document.getElementById('displayEntries')
        .innerText = receipts.length;

}

/* =========================================
   FORM SUBMIT
========================================= */

document
.getElementById('promoForm')
.addEventListener('submit', function() {

    const receipts = [];

    let totalAmount = 0;

    document
    .querySelectorAll('.receipt-block')
    .forEach((block, index) => {

        const amount = parseFloat(

            block.querySelector('.receipt-amount').value || 0

        );

        totalAmount += amount;

        receipts.push({

            receiptNumber: index + 1,

            date:
                block.querySelector('.receipt-date').value,

            time:
                block.querySelector('.receipt-time').value,

            amount: amount,

            store:
                block.querySelector('.store-name').value,

            fileUrl:
                block.querySelector('.file-url').value

        });

    });

    document.getElementById('receiptJSON').value =

        JSON.stringify(receipts);

    document.getElementById('totalAmount').value =

        totalAmount.toFixed(2);

    document.getElementById('entryCount').value =

        receipts.length;

});

/* =========================================
   INITIAL LOAD
========================================= */

addReceipt();
